/* ============================================================================

   ============================================================================ */

function escapeHTML(
    value
) {

    return String(
        value ?? ""
    )
        .replace(
            /&/g,
            "&amp;"
        )
        .replace(
            /</g,
            "&lt;"
        )
        .replace(
            />/g,
            "&gt;"
        )
        .replace(
            /"/g,
            "&quot;"
        )
        .replace(
            /'/g,
            "&#039;"
        );
}


/* ============================================================================
   API
   ============================================================================ */

async function apiFetch(
    path,
    options = {}
) {

    const headers =
        new Headers(
            options.headers || {}
        );

    headers.set(
        "Content-Type",
        "application/json"
    );

    if (sessionToken) {

        headers.set(
            "Authorization",
            "Bearer " +
            sessionToken
        );
    }

    return fetch(
        REGISTRY_API +
        path,
        {
            ...options,
            headers
        }
    );
}

async function handleLogin(
    event
) {

    event.preventDefault();

    const identifier =
        $("loginIdentifier")
            .value
            .trim();

    const password =
        $("loginPassword")
            .value;

    const button =
        $("loginButton");

    const error =
        $("loginError");

    error.style.display =
        "none";

    button.disabled =
        true;

    button.textContent =
        "Signing In...";

    try {

        const response =
            await fetch(
                REGISTRY_API +
                "/api/auth/login",
                {
                    method:
                        "POST",

                    headers: {
                        "Content-Type":
                            "application/json"
                    },

                    body:
                        JSON.stringify({
                            loginIdentifier:
                                identifier,

                            password:
                                password
                        })
                }
            );

        const data =
            await response.json();

        if (
            !response.ok ||
            !data.success ||
            !data.token
        ) {

            throw new Error(
                data.message ||
                data.error ||
                "Authentication failed."
            );
        }

        sessionToken =
            data.token;

        sessionStorage.setItem(
            "suhs_session",
            sessionToken
        );

        applyAccount(
            data
        );

    } catch (errorObject) {

        error.textContent =
            errorObject.message ||
            "Unable to sign in.";

        error.style.display =
            "block";

    } finally {

        button.disabled =
            false;

        button.textContent =
            "Sign In";
    }
}


async function restoreSession() {

    try {

        const response =
            await apiFetch(
                "/api/auth/me"
            );

        if (!response.ok) {
            return false;
        }

        const data =
            await response.json();

        if (
            !data.authenticated
        ) {
            return false;
        }

        applyAccount(
            data
        );

        return true;

    } catch {

        return false;
    }
}

function selectInitialDomain() {

    if (!domains.length) {

        activeDomain =
            null;

        return;
    }

    const wanted =
        String(
            requestedDomain || ""
        )
            .trim()
            .toLowerCase();

    activeDomain =
        wanted
            ? domains.find(
                domain =>
                    String(
                        domain.fqdn || ""
                    )
                        .toLowerCase() ===
                    wanted
            )
            : null;

    if (!activeDomain) {

        activeDomain =
            domains[0];
    }
}


function selectInitialProfile() {

    if (!profiles.length) {

        activeProfile =
            null;

        return;
    }

    const own =
        currentAccount &&
        currentAccount.profile;

    if (own) {

        activeProfile =
            profiles.find(
                profile =>
                    String(profile.id) ===
                    String(own.id)
            ) || null;
    }

    if (!activeProfile) {

        activeProfile =
            profiles[0];
    }
}

async function loadSitePages() {

    if (!activeDomain) {
        return;
    }

    showStatus(
        "Loading website..."
    );

    const response =
        await apiFetch(
            "/site/" +
            encodeURIComponent(
                activeDomain.fqdn
            )
        );

    if (!response.ok) {

        sitePages = [
            {
                content_key:
                    "index.html",

                content:
                    createStarterDocument(
                        activeDomain.fqdn
                    )
            }
        ];

        currentPage =
            "index.html";

        renderPageSelect();

        loadCurrentPage();

        showStatus(
            "New website ready."
        );

        return;
    }

    const data =
        await response.json();

    sitePages =
        Array.isArray(
            data.content
        )
            ? data.content
            : [];

    if (!sitePages.length) {

        sitePages = [
            {
                content_key:
                    "index.html",

                content:
                    createStarterDocument(
                        activeDomain.fqdn
                    )
            }
        ];
    }

    if (
        !sitePages.some(
            page =>
                page.content_key ===
                currentPage
        )
    ) {

        currentPage =
            sitePages[0].content_key;
    }

    renderPageSelect();

    loadCurrentPage();

    showStatus(
        "Website loaded.",
        "success"
    );
}

function injectEditorBridge(
    html
) {

    const bridge = `
<script>
(function(){

    function send(type,payload){

        parent.postMessage(
            {
                suhsEditor:true,
                type:type,
                payload:payload || {}
            },
            "*"
        );
    }

    function pathFor(element){

        const path=[];

        let current=element;

        while(
            current &&
            current.nodeType===1 &&
            current !== document.documentElement
        ){

            let index=0;

            let sibling=current;

            while(
                sibling=sibling.previousElementSibling
            ){
                index++;
            }

            path.unshift(
                current.tagName.toLowerCase() +
                ":" +
                index
            );

            current=current.parentElement;
        }

        return path.join("/");
    }

    function describe(element){

        if(!element){
            return null;
        }

        const r=element.getBoundingClientRect();

        return {
            path:pathFor(element),
            tag:element.tagName.toLowerCase(),
            id:element.id || "",
            className:element.className || "",
            x:r.left,
            y:r.top,
            width:r.width,
            height:r.height,
            text:
                element.childElementCount===0
                    ? element.textContent
                    : "",
            opacity:
                getComputedStyle(element).opacity
        };
    }

    document.addEventListener(
        "click",
        function(event){

            const target=
                event.target;

            send(
                "element-click",
                describe(target)
            );

        },
        true
    );

    document.addEventListener(
        "dblclick",
        function(event){

            const target=
                event.target;

            if(
                target.childElementCount===0 &&
                target.isContentEditable !== true
            ){

                target.contentEditable=true;

                target.focus();

                send(
                    "text-edit",
                    describe(target)
                );
            }

        },
        true
    );

    document.addEventListener(
        "input",
        function(event){

            send(
                "document-input",
                {
                    path:
                        pathFor(
                            event.target
                        )
                }
            );

        },
        true
    );

    send(
        "ready",
        {}
    );

})();
<\/script>`;

    if (
        /<\\/body>/i.test(html)
    ) {

        return html.replace(
            /<\\/body>/i,
            bridge +
            "</body>"
        );

    }

    return html +
        bridge;
}

   ============================================================================ */

window.addEventListener(
    "message",
    event => {

        if (
            !event.data ||
            !event.data.suhsEditor
        ) {
            return;
        }

        const type =
            event.data.type;

        const payload =
            event.data.payload || {};

        if (type === "ready") {

            updateSelectionBox();

            return;
        }

        if (
            type ===
            "element-click"
        ) {

            if (
                activeTool ===
                "select" ||
                activeTool ===
                "text" ||
                activeTool ===
                "pin" ||
                activeTool ===
                "animate"
            ) {

                selectElementByPath(
                    payload.path
                );
            }

            if (
                activeTool ===
                "text"
            ) {

                editSelectedText();
            }

            if (
                activeTool ===
                "pin"
            ) {

                setPinFromElementCenter();
            }

            return;
        }

        if (
            type ===
            "document-input"
        ) {

            syncDocumentFromFrame();

            return;
        }
    }
);


/* ============================================================================
   FRAME DOM
   ============================================================================ */

function getFrameDocument() {

    const frame =
        $("siteFrame");

    return frame.contentDocument ||
        frame.contentWindow.document;
}


function resolveElementPath(
    path
) {

    const doc =
        getFrameDocument();

    if (!doc || !path) {
        return null;
    }

    const parts =
        path.split("/");

    let current =
        doc.documentElement;

    for (
        const part of parts
    ) {

        const pieces =
            part.split(":");

        const tag =
            pieces[0];

        const index =
            Number(
                pieces[1]
            );

        const candidates =
            Array.from(
                current.children
            )
                .filter(
                    child =>
                        child.tagName
                            .toLowerCase() ===
                        tag
                );

        current =
            candidates[index];

        if (!current) {
            return null;
        }
    }

    return current;
}


function selectElementByPath(
    path
) {

    const element =
        resolveElementPath(
            path
        );

    if (!element) {
        return;
    }

    selectedElement =
        element;

    selectedElementPath =
        path;

    updateInspector();

    updateSelectionBox();

    renderLayers();
}

/* ============================================================================
   SELECTION
   ============================================================================ */

function updateSelectionBox() {

    const box =
        $("selectionBox");

    if (
        !box ||
        !selectedElement
    ) {
        if (box) {
            box.style.display =
                "none";
        }
        return;
    }

    if (
        !selectedElement.isConnected
    ) {
        clearSelection();
        return;
    }

    const frame =
        $("siteFrame");

    const canvasFrame =
        $("canvasFrame");

    if (
        !frame ||
        !canvasFrame
    ) {
        return;
    }

    const elementRect =
        selectedElement
            .getBoundingClientRect();

    const frameRect =
        frame.getBoundingClientRect();

    const canvasRect =
        canvasFrame.getBoundingClientRect();

    box.style.display =
        "block";

    box.style.left =
        (
            frameRect.left -
            canvasRect.left +
            elementRect.left -
            frameRect.left
        ) +
        "px";

    box.style.top =
        (
            frameRect.top -
            canvasRect.top +
            elementRect.top -
            frameRect.top
        ) +
        "px";

    box.style.width =
        elementRect.width +
        "px";

    box.style.height =
        elementRect.height +
        "px";

    $("selectionLabel")
        .textContent =
        "<" +
        selectedElement.tagName
            .toLowerCase() +
        ">";
}


function clearSelection() {

    selectedElement =
        null;

    selectedElementPath =
        null;

    $("selectionBox")
        .style.display =
        "none";

    $("selectionInspector")
        .style.display =
        "none";

    $("noSelection")
        .style.display =
        "block";

    renderLayers();
}


/* ============================================================================
   INSPECTOR
   ============================================================================ */

function updateInspector() {

    if (
        !selectedElement
    ) {

        $("selectionInspector")
            .style.display =
            "none";

        $("noSelection")
            .style.display =
            "block";

        return;
    }

    $("selectionInspector")
        .style.display =
        "block";

    $("noSelection")
        .style.display =
        "none";

    const element =
        selectedElement;

    const computed =
        getComputedStyle(
            element
        );

    const rect =
        element.getBoundingClientRect();

    $("selectedTag")
        .value =
        "<" +
        element.tagName
            .toLowerCase() +
        ">";

    $("selectedId")
        .value =
        element.id ||
        "";

    $("selectedClass")
        .value =
        typeof element.className ===
        "string"
            ? element.className
            : "";

    $("posX")
        .value =
        Number(
            parseFloat(
                element.style.left
            )
        ) ||
        Number(
            parseFloat(
                computed.left
            )
        ) ||
        0;

    $("posY")
        .value =
        Number(
            parseFloat(
                element.style.top
            )
        ) ||
        Number(
            parseFloat(
                computed.top
            )
        ) ||
        0;

    $("objWidth")
        .value =
        Math.round(
            rect.width
        );

    $("objHeight")
        .value =
        Math.round(
            rect.height
        );

    const opacity =
        Number(
            computed.opacity
        );

    const safeOpacity =
        Number.isFinite(
            opacity
        )
            ? opacity
            : 1;

    $("objectOpacity")
        .value =
        Math.round(
            safeOpacity *
            100
        );

    $("objectOpacityValue")
        .textContent =
        Math.round(
            safeOpacity *
            100
        ) +
        "%";

    $("selectedText")
        .value =
        element.childElementCount ===
        0
            ? element.textContent ||
              ""
            : "";

    $("fontFamily")
        .value =
        normalizeFontValue(
            computed.fontFamily
        );

    $("fontSize")
        .value =
        parseFloat(
            computed.fontSize
        ) ||
        "";

    $("textColor")
        .value =
        toHex(
            computed.color
        );

    $("backgroundColor")
        .value =
        toHex(
            computed.backgroundColor
        );

    $("backgroundImage")
        .value =
        extractBackgroundImageUrl(
            computed.backgroundImage
        );

    $("backgroundSize")
        .value =
        element.style
            .backgroundSize ||
        computed.backgroundSize ||
        "auto";

    $("backgroundPosition")
        .value =
        element.style
            .backgroundPosition ||
        computed.backgroundPosition ||
        "center";

    $("borderWidth")
        .value =
        parseFloat(
            computed.borderTopWidth
        ) ||
        0;

    $("borderRadius")
        .value =
        parseFloat(
            computed.borderRadius
        ) ||
        0;

    $("borderColor")
        .value =
        toHex(
            computed.borderTopColor
        );

    $("borderStyle")
        .value =
        computed.borderTopStyle ||
        "solid";

    const shadow =
        parseBoxShadow(
            computed.boxShadow
        );

    $("shadowX")
        .value =
        shadow.x;

    $("shadowY")
        .value =
        shadow.y;

    $("shadowBlur")
        .value =
        shadow.blur;

    $("shadowColor")
        .value =
        shadow.color;

    $("shadowOpacity")
        .value =
        shadow.opacity;

    const origin =
        computed.transformOrigin
            .split(" ");

    $("pivotX")
        .value =
        transformOriginToPercent(
            origin[0],
            rect.width
        );

    $("pivotY")
        .value =
        transformOriginToPercent(
            origin[1],
            rect.height
        );

    $("linkUrl")
        .value =
        extractUrl(
            element
        );

    $("linkTarget")
        .value =
        extractTarget(
            element
        );
}

function updateShadow() {

    if (
        !selectedElement
    ) {
        return;
    }

    const x =
        Number(
            $("shadowX")
                .value
        ) || 0;

    const y =
        Number(
            $("shadowY")
                .value
        ) || 0;

    const blur =
        Math.max(
            0,
            Number(
                $("shadowBlur")
                    .value
            ) || 0
        );

    const opacity =
        Math.max(
            0,
            Math.min(
                100,
                Number(
                    $("shadowOpacity")
                        .value
                ) || 0
            )
        ) /
        100;

    const rgb =
        hexToRgb(
            $("shadowColor")
                .value
        );

    if (
        x === 0 &&
        y === 0 &&
        blur === 0
    ) {

        selectedElement.style
            .boxShadow =
            "none";

    } else {

        selectedElement.style
            .boxShadow =
            `${x}px ${y}px ${blur}px rgba(${rgb.r},${rgb.g},${rgb.b},${opacity})`;
    }

    syncDocumentFromFrame();

    saveUndoState();

    updateSelectionBox();
}


function updatePivot() {

    if (
        !selectedElement
    ) {
        return;
    }

    const x =
        Math.max(
            0,
            Math.min(
                100,
                Number(
                    $("pivotX")
                        .value
                ) || 0
            )
        );

    const y =
        Math.max(
            0,
            Math.min(
                100,
                Number(
                    $("pivotY")
                        .value
                ) || 0
            )
        );

    selectedElement.style
        .transformOrigin =
        `${x}% ${y}%`;

    syncDocumentFromFrame();

    saveUndoState();
}

        $("linkTarget")
            .value;

    if (
        selectedElement.tagName ===
        "A"
    ) {

        selectedElement
            .setAttribute(

function nextLayerNumber() {

    const doc =
        getFrameDocument();

    if (!doc) {
        return 1;
    }

    const numbers =
        Array.from(
            doc.querySelectorAll(
                "[data-suhs-editor-object]"
            )
        )
            .map(
                element =>
                    parseInt(
                        getComputedStyle(
                            element
                        ).zIndex,
                        10
                    )
            )
            .filter(
                Number.isFinite
            );

    return numbers.length
        ? Math.max(
            ...numbers
        ) + 1
        : 1;
}

                document.createElement(
                    "div"

function layerName(
    element
) {

    if (
        element.id

            "#" +
            element.id

            element.textContent ||

                /\s+/g,

            " — " +
            text.slice(
                0,

    return element.tagName
        .toLowerCase();
}

function pathForElement(
    element
) {

    if (
        !element ||
        !element.ownerDocument
    ) {
        return null;
    }

    const parts =
        [];

    let current =
        element;

    while (
        current &&
        current.nodeType === 1 &&
        current !==
            current.ownerDocument.documentElement
    ) {

        let index =
            0;

        let sibling =
            current.previousElementSibling;

        while (
            sibling
        ) {

            if (
                sibling.tagName ===
                current.tagName
            ) {
                index++;
            }

            sibling =
                sibling.previousElementSibling;
        }

        parts.unshift(
            current.tagName
                .toLowerCase() +
            ":" +
            index
        );

        current =
            current.parentElement;
    }

    return parts.join("/");
}

function rotateSelected(
    degrees
) {

    if (
        !selectedElement
    ) {
        return;
    }

    saveUndoState();

    const transform =
        selectedElement.style
            .transform ||
        "";

    const parsed =
        parseTransform(
            transform
        );

    const nextRotation =
        parsed.rotation +
        degrees;

    selectedElement.style
        .transform =
        preserveTransforms(
            parsed,
            nextRotation
        );

    syncDocumentFromFrame();

    saveUndoState();

    updateInspector();

    updateSelectionBox();
}


function parseTransform(
    transform
) {

    if (
        !transform ||
        transform ===
        "none"
    ) {

        return {

            rotation:
                0,

            scaleX:
                1,

            scaleY:
                1,

            translateX:
                0,

            translateY:
                0
        };
    }

    const rotate =
        transform.match(
            /rotate\(\s*(-?[\d.]+)deg\s*\)/
        );

    if (
        rotate
    ) {

        return {

            rotation:
                Number(
                    rotate[1]
                ),

            scaleX:
                1,

            scaleY:
                1,

            translateX:
                0,

            translateY:
                0
        };
    }

    const matrix =
        transform.match(
            /matrix\(([^)]+)\)/
        );

    if (
        matrix
    ) {

        const values =
            matrix[1]
                .split(",")
                .map(
                    Number
                );

        if (
            values.length >= 6
        ) {

            const [
                a,
                b,
                c,
                d,
                tx,
                ty
            ] =
                values;

            return {

                rotation:
                    Math.atan2(
                        b,
                        a
                    ) *
                    180 /
                    Math.PI,

                scaleX:
                    Math.sqrt(
                        a * a +
                        b * b
                    ),

                scaleY:
                    Math.sqrt(
                        c * c +
                        d * d
                    ),

                translateX:
                    tx,

                translateY:
                    ty
            };
        }
    }

    return {

        rotation:
            0,

        scaleX:
            1,

        scaleY:
            1,

        translateX:
            0,

        translateY:
            0
    };
}


function preserveTransforms(
    parsed,
    rotation
) {

    return (
        `translate(${parsed.translateX}px,${parsed.translateY}px) ` +
        `scale(${parsed.scaleX},${parsed.scaleY}) ` +
        `rotate(${rotation}deg)`
    );
}

    const doc =
        getFrameDocument();

    if (!doc) {
        return;
    }

    doc.addEventListener(
        "pointerdown",
        startPainting
    );

    doc.addEventListener(
        "pointermove",
        continuePainting
    );

    doc.addEventListener(
        "pointerup",
        stopPainting
    );

    doc.addEventListener(
        "pointercancel",
        stopPainting
    );
}


function disablePaintMode() {

    const doc =
        getFrameDocument();

    if (!doc) {
        return;
    }

    doc.removeEventListener(
        "pointerdown",
        startPainting
    );

    doc.removeEventListener(
        "pointermove",
        continuePainting
    );

    doc.removeEventListener(
        "pointerup",
        stopPainting
    );

    doc.removeEventListener(
        "pointercancel",
        stopPainting
    );

    isPainting =
        false;

    paintPoints =
        [];
}


function startPainting(
    event
) {

    if (
        event.button !==
        0
    ) {
        return;
    }

    isPainting =
        true;

    paintPoints =
        [
            canvasPoint(
                event
            )
        ];

    event.preventDefault();
}


function continuePainting(
    event
) {

    if (
        !isPainting
    ) {
        return;
    }

    paintPoints.push(
        canvasPoint(
            event
        )
    );

    renderPaintPath();
}


function stopPainting() {

    if (
        !isPainting
    ) {
        return;
    }

    isPainting =
        false;

    createPaintSvg();

    paintPoints =
        [];

    syncDocumentFromFrame();

    saveUndoState();
}


function canvasPoint(
    event
) {

    const doc =
        getFrameDocument();

    if (!doc) {

        return {
            x: 0,
            y: 0
        };
    }

    const bodyRect =
        doc.body
            .getBoundingClientRect();

    return {

        x:
            event.clientX -
            bodyRect.left +
            doc.defaultView.scrollX,

        y:
            event.clientY -
            bodyRect.top +
            doc.defaultView.scrollY
    };
}


function renderPaintPath() {

    if (
        paintPoints.length <
        2
    ) {
        return;
    }

    const doc =
        getFrameDocument();

    if (!doc) {
        return;
    }

    let preview =
        doc.getElementById(
            "suhsPaintPreview"
        );

    if (!preview) {

        preview =
            doc.createElementNS(
                "http://www.w3.org/2000/svg",
                "svg"
            );

        preview.id =
            "suhsPaintPreview";

        preview.style.position =
            "absolute";

        preview.style.left =
            "0";

        preview.style.top =
            "0";

        preview.style.width =
            "100%";

        preview.style.height =
            "100%";

        preview.style.pointerEvents =
            "none";

        preview.style.zIndex =
            "9999";

        doc.body.appendChild(
            preview
        );
    }

    preview.innerHTML =
        "";

    const path =
        doc.createElementNS(
            "http://www.w3.org/2000/svg",
            "path"
        );

    path.setAttribute(
        "d",
        paintPoints
            .map(
                (
                    point,
                    index
                ) =>
                    (
                        index ===
                        0
                            ? "M"
                            : "L"
                    ) +
                    point.x +
                    " " +
                    point.y
            )
            .join(" ")
    );

    path.setAttribute(
        "fill",
        "none"
    );

    path.setAttribute(
        "stroke",
        brushColor
    );

    path.setAttribute(
        "stroke-width",
        brushSize
    );

    path.setAttribute(
        "stroke-linecap",
        $("brushTip")
            .value ===
        "square"
            ? "square"
            : "round"
    );

    path.setAttribute(
        "stroke-linejoin",
        "round"
    );

    path.setAttribute(
        "opacity",
        brushOpacity
    );

    preview.appendChild(
        path
    );
}

   ============================================================================ */

function startAnimationRecording() {

    if (
        !selectedElement
    ) {

        showStatus(
            "Select an object first.",
            "error"
        );

        return;
    }

    recordingAnimation =
        true;

    animationStart =
        performance.now();

    recordedAnimations =
        [
            {
                time:
                    0,

                state:
                    captureElementState(
                        selectedElement
                    )
            }
        ];

    $("recordIndicator")
        .classList.add(
            "active"
        );

    $("recordAnimationButton")
        .disabled =
        true;

    $("stopAnimationButton")
        .disabled =
        false;

    showStatus(
        "Animation recording started.",
        "success"
    );
}


function captureElementState(
    element
) {

    const computed =
        getComputedStyle(
            element
        );

    return {

        x:
            parseFloat(
                element.style.left
            ) ||
            parseFloat(
                computed.left
            ) ||
            0,

        y:
            parseFloat(
                element.style.top
            ) ||
            parseFloat(
                computed.top
            ) ||
            0,

        width:
            parseFloat(
                computed.width
            ) ||
            0,

        height:
            parseFloat(
                computed.height
            ) ||
            0,

        opacity:
            parseFloat(
                computed.opacity
            ) ||
            1,

        transform:
            computed.transform ===
            "none"
                ? "none"
                : computed.transform
    };
}


function stopAnimationRecording() {

    if (
        !recordingAnimation ||
        !selectedElement
    ) {
        return;
    }

    recordedAnimations.push(
        {
            time:
                performance.now() -
                animationStart,

            state:
                captureElementState(
                    selectedElement
                )
        }
    );

    recordingAnimation =
        false;

    $("recordIndicator")
        .classList.remove(
            "active"
        );

    $("recordAnimationButton")
        .disabled =
        false;

    $("stopAnimationButton")
        .disabled =
        true;

    installAnimation(
        recordedAnimations
    );

    animationStart =
        null;
}


function installAnimation(
    frames
) {

    if (
        !selectedElement ||
        !frames ||
        frames.length <
        2
    ) {
        return;
    }

    const doc =
        getFrameDocument();

    if (!doc) {
        return;
    }

    const animationId =
        "suhs-animation-" +
        Math.random()
            .toString(36)
            .slice(2);

    const style =
        doc.createElement(
            "style"
        );

    const duration =
        Math.max(
            300,
            Number(
                $("animationDuration")
                    .value
            ) ||
            1000
        );

    const trigger =
        $("animationTrigger")
            .value;

    const easing =
        $("animationEasing")
            .value;

    const keyframes =
        frames
            .map(
                frame => {

                    const percentage =
                        Math.max(
                            0,
                            Math.min(
                                100,
                                (
                                    frame.time /
                                    Math.max(
                                        1,
                                        frames[
                                            frames.length - 1
                                        ].time
                                    )
                                ) *
                                100
                            )
                        );

                    const state =
                        frame.state;

                    return `
                        ${percentage}% {
                            left:${state.x}px;
                            top:${state.y}px;
                            width:${state.width}px;
                            height:${state.height}px;
                            opacity:${state.opacity};
                            transform:${state.transform};
                        }
                    `;
                }
            )
            .join("");

    style.textContent =
        `
            @keyframes ${animationId} {

                ${keyframes}
            }

            [data-suhs-animation="${animationId}"] {
                animation:
                    ${animationId}
                    ${duration}ms
                    ${easing}
                    1
                    both;
            }

            [data-suhs-animation="${animationId}"]:hover {
            }
        `;

    doc.head.appendChild(
        style
    );

    selectedElement.style
        .position =
        "absolute";

    selectedElement.dataset
        .suhsAnimation =
        animationId;

    if (
        trigger ===
        "load"
    ) {

        selectedElement
            .setAttribute(
                "data-suhs-animation",
                animationId
            );

    } else if (
        trigger ===
        "hover"
    ) {

        selectedElement
            .style
            .animation =
            "none";

        const hoverClass =
            "suhs-hover-" +
            animationId;

        const hoverStyle =
            doc.createElement(
                "style"
            );

        hoverStyle.textContent =
            `
                .${hoverClass}:hover {
                    animation:
                        ${animationId}
                        ${duration}ms
                        ${easing}
                        1
                        both;
                }
            `;

        doc.head.appendChild(
            hoverStyle
        );

        selectedElement
            .classList.add(
                hoverClass
            );

    } else if (
        trigger ===
        "click"
    ) {

        selectedElement
            .addEventListener(
                "click",
                function() {

                    this.style
                        .animation =
                        "none";

                    void this.offsetWidth;

                    this.style
                        .animation =
                        `${animationId} ${duration}ms ${easing} 1 both`;
                }
            );

    } else {

        selectedElement
            .setAttribute(
                "data-suhs-animation",
                animationId
            );
    }

    addAnimationToSequence(
        animationId
    );

    renderAnimationTimeline();

    syncDocumentFromFrame();

    saveUndoState();

    showStatus(
        "Animation installed.",
        "success"
    );
}


function addAnimationToSequence(
    name
) {

    if (
        !name
    ) {
        return;
    }

    animationSequence.push(
        name
    );

    renderAnimationTimeline();
}


function installSequence() {

    if (
        !selectedElement ||
        !animationSequence.length
    ) {
        return;
    }

    selectedElement.dataset
        .suhsAnimationSequence =
        JSON.stringify(
            animationSequence
        );

    syncDocumentFromFrame();

    saveUndoState();

    renderAnimationTimeline();
}


function renderAnimationTimeline() {

    const timeline =
        $("animationTimeline");

    timeline.innerHTML =
        "";

    recordedAnimations.forEach(
        (
            frame,
            index
        ) => {

            const row =
                document.createElement(
                    "div"
                );

            row.className =
                "animation-row";

            row.innerHTML =
                `
                    <strong>
                        Keyframe ${index + 1}
                    </strong>

                    <div class="animation-meta">
                        ${Math.round(
                            frame.time
                        )} ms
                    </div>
                `;

            timeline.appendChild(
                row
            );
        }
    );

    animationSequence.forEach(
        (
            name,
            index
        ) => {

            const row =
                document.createElement(
                    "div"
                );

            row.className =
                "animation-row";

            row.innerHTML =
                `
                    <strong>
                        Sequence ${index + 1}
                    </strong>

                    <div class="animation-meta">
                        ${escapeHTML(
                            name
                        )}
                    </div>
                `;

            timeline.appendChild(
                row
            );
        }
    );

    if (
        !recordedAnimations.length &&
        !animationSequence.length
    ) {

        timeline.innerHTML =
            `
                <div
                    style="
                        color:#718096;
                        font-size:10px;
                    "
                >
                    No animation recorded.
                </div>
            `;
    }
}


/* ============================================================================
   SERIALIZATION
   ============================================================================ */

function syncDocumentFromFrame() {

    const doc =
        getFrameDocument();

    if (!doc) {
        return currentHTML;
    }

    doc.querySelectorAll(
        "[contenteditable]"
    )
        .forEach(
            element =>
                element.removeAttribute(
                    "contenteditable"
                )
        );

    const bridgeScripts =
        doc.querySelectorAll(
            "script"
        );

    bridgeScripts.forEach(
        script => {

            if (
                script.textContent
                    .includes(
                        "suhsEditor"
                    )
            ) {

                script.remove();
            }
        }
    );

    const serialized =
        "<!DOCTYPE html>\n" +
        doc.documentElement
            .outerHTML;

    currentHTML =
        serialized;

    const page =
        sitePages.find(
            item =>
                item.content_key ===
                currentPage
        );

    if (
        page
    ) {

        page.content =
            serialized;
    }

    $("sourceEditor")
        .value =
        serialized;

    return serialized;
}


function saveCurrentToMemory() {

    syncDocumentFromFrame();
}

    mode

    if (

        );

    $("htmlModeButton")

        );

}

        $("sourceEditor")
            .value;

/* ============================================================================
   SAVE / PUBLISH / PREVIEW
   ============================================================================ */

async function saveCurrentPage() {

    if (
        !activeDomain
    ) {

        showStatus(
            "No domain is selected.",
            "error"
        );

        return;
    }

    saveCurrentToMemory();

    const page =
        sitePages.find(
            item =>
                item.content_key ===
                currentPage
        );

    if (
        !page
    ) {
        return;
    }

    try {

        showStatus(
            "Saving..."
        );

        const response =
            await apiFetch(
                "/site/update",
                {
                    method:
                        "POST",

                    body:
                        JSON.stringify({
                            fqdn:
                                activeDomain.fqdn,

                            content_key:
                                currentPage,

                            content:
                                page.content
                        })
                }
            );

        if (
            !response.ok
        ) {

            const data =
                await response
                    .json()
                    .catch(
                        () =>
                            ({})
                    );

            throw new Error(
                data.message ||
                data.error ||
                "Unable to save website."
            );
        }

        showStatus(
            "Saved " +
            currentPage +
            ".",
            "success"
        );

    } catch (
        errorObject
    ) {

        showStatus(
            errorObject.message,
            "error"
        );
    }
}


async function publishCurrentPage() {

    if (

/* ============================================================================
   PAGE MANAGEMENT
   ============================================================================ */

async function createPage() {

    const name =
        prompt(
            "New page filename",
            "about.html"
        );

    if (
        !name
    ) {
        return;
    }

    const normalized =
        name.trim();

    if (
        !normalized
    ) {
        return;
    }

    if (
        sitePages.some(
            page =>
                page.content_key ===
                normalized
        )
    ) {

        showStatus(
            "A page with that name already exists.",
            "error"
        );

        return;
    }

    saveCurrentToMemory();

    sitePages.push(
        {
            content_key:
                normalized,

            content:
                createStarterDocument(
                    activeDomain
                        ? activeDomain.fqdn
                        : "SUHS"
                )
        }
    );

    currentPage =
        normalized;

    renderPageSelect();

    loadCurrentPage();

    undoStack =
        [];

    redoStack =
        [];

    showStatus(
        "Created " +
        normalized +
        ".",
        "success"
    );
}


async function renamePage() {

    const oldName =
        currentPage;

    const name =
        prompt(
            "Rename page",
            oldName
        );

    if (
        !name
    ) {
        return;
    }

    const normalized =
        name.trim();

    if (
        !normalized ||
        normalized ===
        oldName
    ) {
        return;
    }

    if (
        sitePages.some(
            page =>
                page.content_key ===
                normalized
        )
    ) {

        showStatus(
            "A page with that name already exists.",
            "error"
        );

        return;
    }

    saveCurrentToMemory();

    const page =
        sitePages.find(
            item =>
                item.content_key ===
                oldName
        );

    if (!page) {
        return;
    }

    page.content_key =
        normalized;

    currentPage =
        normalized;

    renderPageSelect();

    loadCurrentPage();

    showStatus(
        "Page renamed.",
        "success"
    );
}


async function deletePage() {

    if (
        sitePages.length <=
        1
    ) {

        showStatus(
            "The website must contain at least one page.",
            "error"
        );

        return;
    }

    if (
        !confirm(
            "Delete " +
            currentPage +
            "?"
        )
    ) {
        return;
    }

    saveCurrentToMemory();

    sitePages =
        sitePages.filter(
            page =>
                page.content_key !==
                currentPage
        );

    currentPage =
        sitePages[0]
            ? sitePages[0].content_key
            : "index.html";

    renderPageSelect();

    loadCurrentPage();

    undoStack =
        [];

    redoStack =
        [];

    showStatus(
        "Page deleted.",
        "success"
    );
}


/* ============================================================================
   UNDO / REDO
   ============================================================================ */

function saveUndoState() {

    const html =
        syncDocumentFromFrame();

    if (
        !html
    ) {
        return;
    }

    if (
        undoStack.length ===
        0 ||
        undoStack[
            undoStack.length - 1
        ] !== html
    ) {

        undoStack.push(
            html
        );

        if (
            undoStack.length >
            100
        ) {

            undoStack.shift();
        }
    }

    redoStack =
        [];
}


function restoreHtml(
    html
) {

    if (
        !html
    ) {
        return;
    }

    currentHTML =
        html;

    const page =
        sitePages.find(
            item =>
                item.content_key ===
                currentPage
        );

    if (
        page
    ) {

        page.content =
            html;
    }

    $("sourceEditor")
        .value =
        html;

    loadVisualFrame();
}


function undo() {

    saveCurrentToMemory();

    if (
        undoStack.length <
        2
    ) {

        showStatus(
            "Nothing to undo."
        );

        return;
    }

    const current =
        undoStack.pop();

    redoStack.push(
        current
    );

    restoreHtml(
        undoStack[
            undoStack.length - 1
        ]
    );

    showStatus(
        "Undo.",
        "success"
    );
}


function redo() {

    if (
        !redoStack.length
    ) {

        showStatus(
            "Nothing to redo."
        );

        return;
    }

    const next =
        redoStack.pop();

    undoStack.push(
        next
    );

    restoreHtml(
        next
    );

    showStatus(
        "Redo.",
        "success"
    );
}

function extractUrl(
    element
) {

    if (
        !element
    ) {
        return "";
    }

    if (
        element.tagName ===
        "A"
    ) {

        return (
            element.getAttribute(
                "href"
            ) ||
            ""
        );
    }

    if (
        element.dataset &&
        element.dataset.suhsLink
    ) {

        return element.dataset
            .suhsLink;
    }

    const anchor =
        element.closest
            ? element.closest(
                "a"
            )
            : null;

    return anchor
        ? (
            anchor.getAttribute(
                "href"
            ) ||
            ""
        )
        : "";
}


function extractTarget(
    element
) {

    if (
        !element
    ) {
        return "_self";
    }

    if (
        element.tagName ===
        "A"
    ) {

        return (
            element.getAttribute(
                "target"
            ) ||
            "_self"
        );
    }

    return (
        element.dataset &&
        element.dataset.suhsTarget
            ? element.dataset.suhsTarget
            : "_self"
    );
}


function extractBackgroundImageUrl(
    value
) {

    const match =
        String(
            value ||
            ""
        )
            .match(
                /url\(["']?(.*?)["']?\)/
            );

    return match
        ? match[1]
        : "";
}


function parseBoxShadow(
    value
) {

    if (
        !value ||
        value ===
        "none"
    ) {

        return {

            x:
                0,

            y:
                0,

            blur:
                0,

            color:
                "#000000",

            opacity:
                0
        };
    }

    let color =
        "#000000";

    let opacity =
        1;

    const rgbaMatch =
        String(
            value
        )
            .match(
                /rgba?\(([^)]+)\)/
            );

    if (
        rgbaMatch
    ) {

        const values =
            rgbaMatch[1]
                .split(",")
                .map(
                    value =>
                        value.trim()
                );

        const r =
            Number(
                values[0]
            ) || 0;

        const g =
            Number(
                values[1]
            ) || 0;

        const b =
            Number(
                values[2]
            ) || 0;

        opacity =
            values.length >= 4
                ? Number(
                    values[3]
                ) || 0
                : 1;

        color =
            "#" +
            [r,g,b]
                .map(
                    value =>
                        value
                            .toString(
                                16
                            )
                            .padStart(
                                2,
                                "0"
                            )
                )
                .join("");
    }

    const lengths =
        value.match(
            /-?\d+(?:\.\d+)?px/g
        ) ||
        [];

    return {

        x:
            parseFloat(
                lengths[0]
            ) || 0,

        y:
            parseFloat(
                lengths[1]
            ) || 0,

        blur:
            parseFloat(
                lengths[2]
            ) || 0,

        color:
            color,

        opacity:
            Math.round(
                opacity *
                100
            )
    };
}


function transformOriginToPercent(
    value,
    size
) {

    if (
        String(
            value
        ).includes(
            "%"
        )
    ) {

        return (
            parseFloat(
                value
            ) || 50
        );
    }

    const pixels =
        parseFloat(
            value
        );

    if (
        !Number.isFinite(
            pixels
        ) ||
        !Number.isFinite(
            size
        ) ||
        size <= 0
    ) {

        return 50;
    }

    return Math.max(
        0,
        Math.min(
            100,
            (
                pixels /
                size
            ) *
            100
        )
    );
}


function normalizeFontValue(
    value
) {

    const first =
        String(
            value ||
            ""
        )
            .split(",")[0]
            .trim()
            .replace(
                /^["']|["']$/g,
                ""
            );

    const known = [

        "Arial",

        "Georgia",

        "Verdana",

        "Tahoma",

        "Courier New"
    ];

    return known.includes(
        first
    )
        ? first
        : "";
}


function toHex(
    value
) {

    const text =
        String(
            value ||
            ""
        )
            .trim();

    if (
        /^#[0-9a-f]{6}$/i
            .test(
                text
            )
    ) {

        return text;
    }

    const rgb =
        text.match(
            /^rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/
        );

    if (
        rgb
    ) {

        return (
            "#" +
            [
                rgb[1],
                rgb[2],
                rgb[3]
            ]
                .map(
                    value =>
                        Number(
                            value
                        )
                            .toString(
                                16
                            )
                            .padStart(
                                2,
                                "0"
                            )
                )
                .join("")
        );
    }

    return "#000000";
}


function hexToRgb(
    value
) {

    const hex =
        String(
            value ||
            ""
        )
            .replace(
                "#",
                ""
            );

    if (
        hex.length ===
        3
    ) {

        return {

            r:
                parseInt(
                    hex[0] +
                    hex[0],
                    16
                ),

            g:
                parseInt(
                    hex[1] +
                    hex[1],
                    16
                ),

            b:
                parseInt(
                    hex[2] +
                    hex[2],
                    16
                )
        };
    }

    if (
        hex.length ===
        6
    ) {

        return {

            r:
                parseInt(
                    hex.slice(
                        0,
                        2
                    ),
                    16
                ),

            g:
                parseInt(
                    hex.slice(
                        2,
                        4
                    ),
                    16
                ),

            b:
                parseInt(
                    hex.slice(
                        4,
                        6
                    ),
                    16
                )
        };
    }

    return {

        r:
            0,

        g:
            0,

        b:
            0
    };
}


function formatDate(
    value
) {

    if (
        !value
    ) {
        return "";
    }

    const numeric =
        Number(
            value
        );

    if (
        Number.isFinite(
            numeric
        )
    ) {

        const milliseconds =
            numeric <
            100000000000
                ? numeric * 1000
                : numeric;

        return new Date(
            milliseconds
        ).toLocaleString();
    }

    const date =
        new Date(
            value
        );

    return Number.isNaN(
        date.getTime()
    )
        ? String(value)
        : date.toLocaleString();
}

/* ============================================================================
   FRAME DRAGGING
   ============================================================================ */

let dragState =
    null;


function frameMouseDown(
    event
) {

    if (
        activeTool !==
        "animate"
    ) {
        return;
    }

    if (
        !selectedElement
    ) {
        return;
    }

    if (
        event.target !==
        selectedElement
    ) {
        return;
    }

    if (
        event.button !==
        0
    ) {
        return;
    }

    event.preventDefault();

    const computed =
        getComputedStyle(
            selectedElement
        );

    if (
        computed.position ===
        "static"
    ) {

        selectedElement.style
            .position =
            "absolute";
    }

    const currentLeft =
        parseFloat(
            selectedElement.style.left ||
            computed.left
        );

    const currentTop =
        parseFloat(
            selectedElement.style.top ||
            computed.top
        );

    dragState = {

        startX:
            event.clientX,

        startY:
            event.clientY,

        startLeft:
            Number.isFinite(
                currentLeft
            )
                ? currentLeft
                : 0,

        startTop:
            Number.isFinite(
                currentTop
            )
                ? currentTop
                : 0
    };

    saveUndoState();
}


function frameMouseMove(
    event
) {

    if (
        !dragState ||
        !selectedElement
    ) {
        return;
    }

    const dx =
        event.clientX -
        dragState.startX;

    const dy =
        event.clientY -
        dragState.startY;

    let x =
        dragState.startLeft +
        dx;

    let y =
        dragState.startTop +
        dy;

    if (
        snapMode ===
        "grid" ||
        snapMode ===
        "all"
    ) {

        x =
            snapValue(
                x
            );

        y =
            snapValue(
                y
            );
    }

    selectedElement.style
        .left =
        x +
        "px";

    selectedElement.style
        .top =
        y +
        "px";

    if (
        recordingAnimation
    ) {

        recordedAnimations.push(
            {
                time:
                    performance.now() -
                    animationStart,

                state:
                    captureElementState(
                        selectedElement
                    )
            }
        );
    }

    updateInspector();

    updateSelectionBox();
}


function frameMouseUp() {

    if (
        !dragState
    ) {
        return;
    }

    dragState =
        null;

    syncDocumentFromFrame();

    saveUndoState();

    updateSelectionBox();
}


document.addEventListener(
    "mousemove",
    frameMouseMove
);

document.addEventListener(
    "mouseup",
    frameMouseUp
);


/* ============================================================================
   SNAP
   ============================================================================ */

function snapValue(
    value
) {

    if (
        snapMode ===
        "none"
    ) {
        return value;
    }

    const size =
        Math.max(
            1,
            Number(
                gridSize
            ) || 20
        );

    if (
        snapMode ===
        "grid" ||
        snapMode ===
        "all"
    ) {

        return Math.round(
            value /
            size
        ) *
        size;
    }

    return value;
}

/* ============================================================================
   KEYBOARD SHORTCUTS
   ============================================================================ */

document.addEventListener(
    "keydown",
    event => {

        const command =
            event.ctrlKey ||
            event.metaKey;

        if (
            command &&
            event.key.toLowerCase() ===
            "s"
        ) {

            event.preventDefault();

            saveCurrentPage();

            return;
        }

        if (
            command &&
            event.key.toLowerCase() ===
            "z"
        ) {

            event.preventDefault();

            if (
                event.shiftKey
            ) {

                redo();

            } else {

                undo();
            }

            return;
        }

        if (
            command &&
            event.key.toLowerCase() ===
            "y"
        ) {

            event.preventDefault();

            redo();

            return;
        }

        if (
            event.key ===
            "Escape"
        ) {

            closePanels();

            if (
                $("composeModal")
                    .style
                    .display ===
                "flex"
            ) {

                $("composeModal")
                    .style
                    .display =
                    "none";
            }

            return;
        }

        if (
            event.key !==
            "Delete"
        ) {
            return;
        }

        const target =
            event.target;

        if (
            target instanceof
                HTMLInputElement ||
            target instanceof
                HTMLTextAreaElement ||
            target instanceof
                HTMLSelectElement
        ) {
            return;
        }

        if (
            selectedElement
        ) {

            deleteSelected();
        }
    }
);


/* ============================================================================
   FRAME LOAD / BINDING
   ============================================================================ */

function bindFrame() {

    const doc =
        getFrameDocument();

    if (!doc) {
        return;
    }

    if (
        doc.documentElement.dataset
            .suhsBound ===
        "1"
    ) {
        return;
    }

    doc.documentElement.dataset
        .suhsBound =
        "1";

    doc.addEventListener(
        "click",
        frameDocumentClick,
        true
    );

    doc.addEventListener(
        "mousedown",
        frameMouseDown,
        true
    );

    doc.addEventListener(
        "input",
        () => {

            sourceDirty =
                true;

            syncDocumentFromFrame();

        },
        true
    );
}


function frameDocumentClick(
    event
) {

    if (
        activeTool ===
        "paint"
    ) {
        return;
    }

    const target =
        event.target;

    if (
        !target ||
        target ===
        target.ownerDocument.body ||
        target ===
        target.ownerDocument.documentElement
    ) {
        return;
    }

    event.preventDefault();

    event.stopPropagation();

    const object =
        target.closest
            ? target.closest(
                "[data-suhs-editor-object]"
            )
            : null;

    const selected =
        object ||
        target;

    selectedElement =
        selected;

    selectedElement.dataset
        .suhsEditorObject =
        "1";

    selectedElementPath =
        pathForElement(
            selectedElement
        );

    updateInspector();

    updateSelectionBox();

    renderLayers();

    if (
        activeTool ===
        "text"
    ) {

        editSelectedText();
    }

    if (
        activeTool ===
        "pin"
    ) {

        setPinFromElementCenter();
    }
}
